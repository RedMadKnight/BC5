+// The pool's free list (g_amm_auto_pool_mutex held): insert a range and merge its neighbours.
+static void AmmAutoFreeLocked(uint64_t start, uint64_t size) {
+	if (size == 0) {

+	auto next = g_amm_auto_free.lower_bound(start);
+	if (next != g_amm_auto_free.begin()) {
+		auto prev = std::prev(next);

+	}

+// BC5 (experiment 0203): KYTY_BC5_APR_READAHEAD=<n> (1-16, default 4) gives each priority's queue n reader
+// threads beside its worker. In the second title's cutscene bursts (80-125 submissions a second) one serial worker
+// per priority fell 3-16 s behind, and the cutscene clock voice's next read came late or never (exps 0200, 0203).

+// worker still takes the buffers in order and runs every command at its turn; at a read it copies the host data

+// Reads [file_offset, file_offset + size) of the host file into *out, as ReadHostFileToGuest would read it
+// (nothing past the end of the file; OK with no data at or past the end), without touching guest memory.
+static int ReadHostFileToHost(const std::string& host_path, uint64_t file_offset, uint64_t size, std::vector<uint8_t>* out) {
+	out->clear();
+	if (size == 0) {
+		return OK;
+	}
+	Common::File file;
+	if (!file.Open(host_path, Common::File::Mode::Read)) {
+		return LibKernel::KERNEL_ERROR_ENOENT;
+	}
+	const auto file_size = file.Size();
+	if (file_offset >= file_size) {
+		file.Close();
+		return OK;
+	}
+	if (!file.Seek(file_offset)) {
+		file.Close();
+		return LibKernel::KERNEL_ERROR_EINVAL;
+	}
+	const auto readable = std::min<uint64_t>(size, file_size - file_offset);
+	out->resize(static_cast<size_t>(readable));
+	uint64_t done = 0;
+	while (done < readable) {
+		const auto request = static_cast<uint32_t>(std::min<uint64_t>(APR_HOST_READ_CHUNK_SIZE, readable - done));
+		uint32_t   read    = 0;
+		file.Read(out->data() + done, request, &read);
+		if (read == 0) {
+			break;
+		}
+		done += read;
+	}
+	out->resize(static_cast<size_t>(done));
+	file.Close();
+	return OK;
+}

+static void AprReaderThread(size_t qi, size_t k) {
