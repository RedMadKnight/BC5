# BC5 experiment 0384: the shadow prototype and its tests, on their own. It compiles backend/src/full_state.cpp unmodified
# beside them; nothing here is part of the backend's build.
cmake_minimum_required(VERSION 3.25)
project(bc5_exp0384 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Before Catch2: everything linked together has to be built with it.
option(BC5_0384_ASAN "Build with AddressSanitizer (MSVC)" OFF)
if(BC5_0384_ASAN)
    add_compile_options(/fsanitize=address /Zi)
    add_link_options(/INCREMENTAL:NO)
endif()
find_package(Catch2 3 QUIET)
if(NOT Catch2_FOUND)
    include(FetchContent)
    FetchContent_Declare(Catch2 GIT_REPOSITORY https://github.com/catchorg/Catch2.git GIT_TAG v3.7.1)
    FetchContent_MakeAvailable(Catch2)
endif()
add_executable(shadow0384 shadow_test.cpp ../../backend/src/full_state.cpp)
target_include_directories(shadow0384 PRIVATE ../../backend/include .)
target_link_libraries(shadow0384 PRIVATE Catch2::Catch2WithMain)
if(MSVC)
    target_compile_options(shadow0384 PRIVATE /W4 /WX)
else()
    target_compile_options(shadow0384 PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wsign-conversion -Werror)
endif()
